% Worked example of the line-opening screen on a four-bus network (DC power flow; reactance 1 on
% AB, DC and BD, 2 on BC and AD; 100 MW injected at A and withdrawn at C; ratings AB 70, BC 60,
% AD 60, DC 55 MW). Opening BD moves its 20 MW onto the two paths with distribution factors
% -0.5 (AB, DC) and +0.5 (BC, AD), which relieves the overloaded line DC.
\begin{tikzpicture}

    \draw[draw=ink, line width=1.2pt] (B)--(D);
  \fi
  \foreach \p/\q in {A/B,B/C,A/D,D/C}{\draw[draw=ink, line width=1.2pt] (\p)--(\q);}

  \node[font=\scriptsize\bfseries, text=ink, left=1mm] at (A) {A};
  \node[font=\scriptsize\bfseries, text=ink, right=1mm] at (C) {C};
