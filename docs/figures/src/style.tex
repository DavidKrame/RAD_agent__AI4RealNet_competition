% Numbered badge of a stage of act(): the same numbers are used in every figure.
\newcommand{\stg}[1]{\tikz[baseline=(b.base)]\node[circle, fill=ink, text=paper, inner sep=0pt,
  minimum size=3.8mm, font=\scriptsize\bfseries] (b) {#1};}

% Locator strip placed at the top of a zoomed figure: the four stages of act() with the zoomed
% stage highlighted, followed by the path inside that stage.  Usage: \stagemap{x,y}{k}{path}.
\newcommand{\stagemap}[3]{%
  \begin{scope}[shift={(#1)}]
  \foreach \i/\n in {1/look-ahead,2/withholding,3/reactive control,4/safeguard}{%
    \ifnum\i=#2\relax\def\lf{hl}\def\ld{hledge}\def\lw{1.1pt}\else\def\lf{grey}\def\ld{edge}\def\lw{0.4pt}\fi
    \node[draw=\ld, fill=\lf, line width=\lw, rounded corners=1.5pt, minimum width=22mm,
          minimum height=5mm, inner sep=1pt, font=\scriptsize, text=ink] (loc\i)
          at ({(\i-1)*2.65},0) {\textbf{\i}\ \n};}
  \foreach \i/\j in {1/2,2/3,3/4}{\draw[-{Latex[length=1.2mm]}, draw=edge, thin] (loc\i)--(loc\j);}
  \node[anchor=west, font=\scriptsize\bfseries, text=hledge] at ($(loc4.east)+(3mm,0)$) {#3};
  \end{scope}}

% Same strip for the three time scales of the overview.  Usage: \lanemap{x,y}{k}{path}.
\newcommand{\lanemap}[3]{%
  \begin{scope}[shift={(#1)}]
  \foreach \i/\n in {1/offline,2/train(),3/act()}{%
    \ifnum\i=#2\relax\def\lf{hl}\def\ld{hledge}\def\lw{1.1pt}\else\def\lf{grey}\def\ld{edge}\def\lw{0.4pt}\fi
    \node[draw=\ld, fill=\lf, line width=\lw, rounded corners=1.5pt, minimum width=19mm,
          minimum height=5mm, inner sep=1pt, font=\scriptsize, text=ink] (lan\i)
          at ({(\i-1)*2.5},0) {\n};}
  \foreach \i/\j in {1/2,2/3}{\draw[-{Latex[length=1.2mm]}, draw=edge, thin] (lan\i)--(lan\j);}
  \node[anchor=west, font=\scriptsize\bfseries, text=hledge] at ($(lan3.east)+(3mm,0)$) {#3};
  \end{scope}}
